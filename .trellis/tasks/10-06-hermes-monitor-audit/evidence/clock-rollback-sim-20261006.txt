# -*- coding: utf-8 -*-
# 10-06-hermes-monitor-audit 沙盘补测 Sim1:时钟回拨/未来戳(单元级注入)
from __future__ import annotations

import os
import socket
import subprocess
import sys
from datetime import datetime, timedelta
from pathlib import Path
from zoneinfo import ZoneInfo

import pytest

from myssia.cron.constants import FIRE_CLAIM_TTL_SECONDS
from myssia.cron.jobs import CronJobs, claim_is_live

TZ = ZoneInfo("Asia/Shanghai")
BASE = datetime(2026, 10, 6, 14, 0, tzinfo=TZ)
LIVE_OWNER = f"{socket.gethostname()}:{os.getpid()}"


class Clock:
    def __init__(self, at: datetime) -> None:
        self.now = at

    def __call__(self) -> datetime:
        return self.now

    def tick(self, **delta: float) -> None:
        self.now = self.now + timedelta(**delta)


@pytest.fixture()
def clock() -> Clock:
    return Clock(BASE)


@pytest.fixture()
def jobs(tmp_path: Path, clock: Clock) -> CronJobs:
    return CronJobs(tmp_path, now_fn=clock)


def raw_of(jobs_api: CronJobs, job_id: str) -> dict:
    return next(j for j in jobs_api.store.load_jobs() if j.get("id") == job_id)


@pytest.fixture()
def dead_pid() -> int:
    child = subprocess.Popen([sys.executable, "-c", "import time; time.sleep(60)"])
    child.kill()
    child.wait()
    return child.pid


def test_sim1_claim_is_live_age_matrix(clock: Clock) -> None:
    claim = {"at": BASE.isoformat(), "by": LIVE_OWNER}
    cases = [
        (timedelta(seconds=100), True),
        (timedelta(seconds=-240), True),
        (timedelta(seconds=-300), True),
        (timedelta(seconds=-300.5), False),
        (timedelta(seconds=-3600), False),
        (timedelta(seconds=301), False),
    ]
    for delta, expected in cases:
        clock.now = BASE + delta
        got = claim_is_live(claim, clock.now, FIRE_CLAIM_TTL_SECONDS)
        assert got is expected, f"elapsed={delta}: claim_is_live={got}, want {expected}"


def test_sim1_rollback_within_tolerance_blocks_reclaim(
    jobs: CronJobs, clock: Clock
) -> None:
    job = jobs.create_job("plugins/news.yaml", "every 5m")
    assert jobs.claim_job_for_fire(job["id"]) is True
    clock.tick(minutes=-4)
    assert jobs.claim_job_for_fire(job["id"]) is False


def test_sim1_beyond_tolerance_reclaimable(jobs: CronJobs, clock: Clock) -> None:
    job = jobs.create_job("plugins/news.yaml", "every 5m")
    jobs.claim_job_for_fire(job["id"])
    clock.now = BASE + timedelta(seconds=-301)
    assert jobs.claim_job_for_fire(job["id"]) is True


def test_sim1_far_future_stamp_reclaimable_even_live_owner(jobs: CronJobs) -> None:
    job = jobs.create_job("plugins/news.yaml", "every 5m")
    raw = raw_of(jobs, job["id"])
    raw["fire_claim"] = {
        "at": (BASE + timedelta(hours=1)).isoformat(),
        "by": LIVE_OWNER,
    }
    with jobs.store.jobs_lock():
        jobs.store.save_jobs([raw], replace=True)
    assert jobs.claim_job_for_fire(job["id"]) is True


def test_sim1_rollback_does_not_save_dead_owner(
    jobs: CronJobs, clock: Clock, dead_pid: int
) -> None:
    job = jobs.create_job("plugins/news.yaml", "every 5m")
    raw = raw_of(jobs, job["id"])
    raw["fire_claim"] = {
        "at": BASE.isoformat(),
        "by": f"{socket.gethostname()}:{dead_pid}",
    }
    with jobs.store.jobs_lock():
        jobs.store.save_jobs([raw], replace=True)
    clock.tick(minutes=-4)
    assert jobs.claim_job_for_fire(job["id"]) is True


def test_sim1_heartbeat_survives_rollback(jobs: CronJobs, clock: Clock) -> None:
    job = jobs.create_job("plugins/news.yaml", "every 5m")
    jobs.claim_job_for_fire(job["id"])
    owner = raw_of(jobs, job["id"])["fire_claim"]["by"]
    clock.tick(minutes=-2)
    assert jobs.heartbeat_fire_claim(job["id"], expected_owner=owner) is True
    assert jobs.heartbeat_fire_claim(job["id"], expected_owner="not-me:1:x") is False
