# holder.py(v2;v1 缺信封处理即首跑事故根因)
import fcntl, json, os, sys, time
lock_path, hold_seconds, jobs_path, target_id = (
    sys.argv[1], float(sys.argv[2]), sys.argv[3], sys.argv[4])
fd = open(lock_path, "a+")
fcntl.flock(fd, fcntl.LOCK_EX)
print(f"HELD epoch={time.time():.3f}", flush=True)
time.sleep(10.0)  # 让 CLI 先进入等锁窗口
with open(jobs_path, encoding="utf-8") as f:
    doc = json.load(f)
jobs = doc["jobs"] if isinstance(doc, dict) else doc
hit = False
for job in jobs:
    if job.get("id") == target_id:
        job["name"] = "alpha-renamed-by-holder"
        hit = True
with open(jobs_path, "w", encoding="utf-8") as f:
    json.dump(doc, f, ensure_ascii=False, indent=2)
    f.flush(); os.fsync(f.fileno())
print(f"HOLDER_WROTE epoch={time.time():.3f} target_found={hit}", flush=True)
time.sleep(max(0.0, hold_seconds - 10.0))
fcntl.flock(fd, fcntl.LOCK_UN)
print(f"RELEASED epoch={time.time():.3f}", flush=True)
